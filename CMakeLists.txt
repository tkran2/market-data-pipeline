cmake_minimum_required(VERSION 3.20)
project(market_data_pipeline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MDPIPE_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(mdp_options INTERFACE)
target_include_directories(mdp_options INTERFACE include)
target_compile_options(mdp_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(MDPIPE_SANITIZE)
  target_compile_options(mdp_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mdp_options INTERFACE -fsanitize=address,undefined)
endif()
add_executable(book-core src/main.cpp)
target_link_libraries(book-core PRIVATE mdp_options)
add_executable(core-bench src/benchmark.cpp)
target_link_libraries(core-bench PRIVATE mdp_options)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME pipeline_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
set_tests_properties(pipeline_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  ENVIRONMENT "MDPIPE_CORE=$<TARGET_FILE:book-core>")
